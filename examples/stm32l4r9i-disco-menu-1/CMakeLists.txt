cmake_minimum_required(VERSION 3.25)
project(L4R9Menu LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# The same application/preset grows when Generate UI creates its first export.
file(GLOB ui_exports CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/Home/component.json")
file(GLOB menu_handlers CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/handlers/Home.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/handlers/Settings.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/handlers/Storage.hpp")
file(GLOB volume_handlers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/handlers/Volume.hpp")
file(GLOB playback_controls CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/playback_controls.hpp")
set(menu_has_playback 0)
if(playback_controls)
    if(NOT menu_handlers)
        message(FATAL_ERROR "Connect Home handlers before the Playback lesson.")
    endif()
    set(menu_has_playback 1)
endif()
file(GLOB load_handlers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/handlers/Load.hpp")
set(menu_has_load 0)
if(load_handlers)
    if(NOT menu_handlers)
        message(FATAL_ERROR "Connect the three-page menu before Load.")
    endif()
    set(menu_has_load 1)
endif()
set(menu_has_volume 0)
if(volume_handlers)
    if(NOT menu_handlers)
        message(FATAL_ERROR "Connect the three-page menu before the Volume lesson.")
    endif()
    set(menu_has_volume 1)
endif()
set(menu_has_ui 0)
set(menu_has_navigation 0)
if(ui_exports)
    set(menu_has_ui 1)
endif()
if(menu_handlers)
    set(menu_has_navigation 1)
    if(NOT menu_has_ui)
        message(FATAL_ERROR "Generate Home before adding navigation handlers.")
    endif()
endif()
if(menu_has_ui)
    if(menu_has_navigation)
        set(menu_pages Home Settings Storage)
        if(menu_has_load)
            list(APPEND menu_pages Load)
        endif()
        if(menu_has_volume)
            list(APPEND menu_pages Volume)
        endif()
    else()
        set(menu_pages Home)
    endif()
    set(form_aliases "// Generated application aliases; do not edit.\n#pragma once\n")
    foreach(page IN LISTS menu_pages)
        set(output "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/${page}")
        foreach(required IN ITEMS component.json form.hpp display.hpp resources.hpp resources.cmake product_config.hpp display-definition.json logical-input.json)
            if(NOT EXISTS "${output}/${required}")
                message(FATAL_ERROR "Generate UI for ${page}: missing ${required}")
            endif()
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${output}/${required}")
        endforeach()
        file(READ "${output}/component.json" manifest)
        string(JSON status GET "${manifest}" generationStatus)
        if(NOT status STREQUAL "complete")
            message(FATAL_ERROR "Incomplete ${page} export. Generate UI again.")
        endif()
        if(menu_has_navigation)
            if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/handlers/${page}.hpp")
                message(FATAL_ERROR "Install every enabled page handler. See the README handler lessons.")
            endif()
            file(READ "${output}/product_config.hpp" config)
            if(NOT config MATCHES "using Form = lithe::generated::([A-Za-z_][A-Za-z0-9_]*);")
                message(FATAL_ERROR "Unrecognized generated form alias for ${page}.")
            endif()
            set(form_type "${CMAKE_MATCH_1}")
            set(alias "${page}")
            if(page STREQUAL "UI")
                set(alias Home)
            endif()
            string(APPEND form_aliases "#include \"Tasks/UI/Generated/${page}/form.hpp\"\n#include \"Tasks/UI/Generated/${page}/resources.hpp\"\nnamespace menu::forms { using ${alias} = lithe::generated::${form_type}; using ${alias}Resources = lithe::generated::assets::${form_type}::SelectedResources; }\n")
            foreach(contract IN ITEMS display-definition.json logical-input.json)
                file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/Home/${contract}" home_hash)
                file(SHA256 "${output}/${contract}" page_hash)
                if(NOT home_hash STREQUAL page_hash)
                    message(FATAL_ERROR "${page} differs from Home's display/input. Regenerate every screen after configuring the shared UI project.")
                endif()
            endforeach()
        endif()
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/menu_form_types.hpp" "${form_aliases}")
endif()
# The tutorial advances by adding source files; its two platform presets stay fixed.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/menu_features.hpp"
    "#pragma once\nnamespace menu::features { inline constexpr bool hasUi{${menu_has_ui}}, hasNavigation{${menu_has_navigation}}, hasVolume{${menu_has_volume}}, hasPlayback{${menu_has_playback}}, hasLoad{${menu_has_load}}; }\n")
if(menu_has_navigation)
    set(composition "#pragma once\n#include \"menu_features.hpp\"\n#include \"menu_form_types.hpp\"\n#include \"Application/unused_page.hpp\"\n")
    foreach(page IN LISTS menu_pages)
        string(APPEND composition "#include \"Tasks/UI/handlers/${page}.hpp\"\n")
    endforeach()
    string(APPEND composition "namespace menu { struct PageComposition { static constexpr bool hasVolume = features::hasVolume, hasPlayback = features::hasPlayback, hasLoad = features::hasLoad;\n")
    foreach(page IN ITEMS Home Settings Storage Volume Load)
        if(page IN_LIST menu_pages)
            string(APPEND composition "using ${page} = ${page}MenuForm; using ${page}Resources = forms::${page}Resources;\n")
        else()
            string(APPEND composition "using ${page} = UnusedPage; using ${page}Resources = UnusedPage;\n")
        endif()
    endforeach()
    string(APPEND composition "}; }\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/menu_composition.hpp" "${composition}")
    set(menu_implementation "Application/navigation_menu.hpp")
else()
    set(menu_implementation "Application/single_form_menu.hpp")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/menu_selected.hpp" "#pragma once\n#include \"${menu_implementation}\"\n")
message(STATUS "Application: UI=${menu_has_ui}, navigation=${menu_has_navigation}")
set(STARTER_PLATFORM Win32 CACHE STRING "Win32 or STM32")
if(NOT STARTER_PLATFORM MATCHES "^(Win32|STM32)$")
    message(FATAL_ERROR "Select Win32 or STM32")
endif()
add_library(menu_application INTERFACE)
target_include_directories(menu_application INTERFACE ${CMAKE_CURRENT_SOURCE_DIR} "${CMAKE_CURRENT_BINARY_DIR}/generated")

target_compile_options(menu_application INTERFACE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wconversion;-Wshadow;-Wold-style-cast;-Wdouble-promotion;-Werror;-fno-exceptions;-fno-rtti>)
if(menu_has_ui)
    foreach(option IN ITEMS BUILD_RENDERING_FIXTURES BUILD_RENDERING_TESTS BUILD_WIN32 BUILD_TESTS BUILD_DESIGNER_PREVIEW BUILD_STM32U545_PROBE ENABLE_EMBEDDED_BUDGET_GATE)
        set(LITHEVIEW_${option} OFF CACHE BOOL "Application uses Core only" FORCE)
    endforeach()
    set(LITHEVIEW_DESTINATION_FORMAT RGB888 CACHE STRING "Native round display storage" FORCE)
    add_subdirectory(Middlewares/Third_Party/LiTheView EXCLUDE_FROM_ALL)
    target_link_libraries(menu_application INTERFACE LitheView::Core LitheView::CompactFont)
endif()
set(RTOS ${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/FreeRTOS/Source)
set(CMSIS ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS_6/CMSIS)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "CMSIS-RTOS2 requires the i686 Win32 preset")
endif()
if(STARTER_PLATFORM STREQUAL "Win32")
    set(PORT ${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/FreeRTOS-Kernel/portable/MSVC-MingW)
else()
    set(PORT ${RTOS}/portable/GCC/ARM_CM4F)
endif()
add_library(starter_config INTERFACE)
target_include_directories(starter_config INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Platform/${STARTER_PLATFORM} ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(starter_config SYSTEM INTERFACE
    ${RTOS}/include ${RTOS}/CMSIS_RTOS_V2 ${CMSIS}/RTOS2/Include ${PORT})
# Both builds compile the same pinned kernel and CMSIS wrapper. Only port.c changes.
add_library(starter_rtos STATIC
    ${RTOS}/tasks.c ${RTOS}/queue.c ${RTOS}/list.c ${RTOS}/timers.c
    ${RTOS}/event_groups.c ${RTOS}/stream_buffer.c
    ${RTOS}/CMSIS_RTOS_V2/cmsis_os2.c ${PORT}/port.c)
target_link_libraries(starter_rtos PUBLIC starter_config)
add_library(starter_application OBJECT Application/app_main.cpp Tasks/kernel_memory.cpp)
target_link_libraries(starter_application PUBLIC starter_config menu_application)
if(menu_has_ui)
    foreach(page IN LISTS menu_pages)
        include(Tasks/UI/Generated/${page}/resources.cmake)
        target_sources(starter_application PRIVATE ${LITHEVIEW_SELECTED_RESOURCE_SOURCES})
    endforeach()
endif()
if(STARTER_PLATFORM STREQUAL "Win32")
    add_executable(starter WIN32 Platform/Win32/main.cpp)
    include(cmake/front_panel.cmake)
    if(menu_has_ui)
        target_sources(starter_application PRIVATE Platform/Win32/ui.cpp Platform/Win32/display.cpp Platform/Win32/input.cpp)
    else()
        target_sources(starter_application PRIVATE Application/no_ui.cpp Platform/Win32/no_display.cpp)
    endif()
    target_link_libraries(starter PRIVATE starter_application starter_rtos user32 gdi32 winmm)
    target_link_options(starter PRIVATE -static -Wl,--gc-sections)
    enable_testing()
    add_test(NAME one_application_task COMMAND starter --self-test)
    set_tests_properties(one_application_task PROPERTIES TIMEOUT 20 PASS_REGULAR_EXPRESSION "PASS:")
    if(menu_has_navigation)
        add_executable(starter_input_checks Tests/menu.cpp)
        target_link_libraries(starter_input_checks PRIVATE menu_application)
        target_link_options(starter_input_checks PRIVATE -static)
        add_test(NAME menu_navigation_contract COMMAND starter_input_checks)
        if(menu_has_volume)
            add_executable(volume_checks Tests/volume.cpp)
            target_link_libraries(volume_checks PRIVATE menu_application)
            target_link_options(volume_checks PRIVATE -static)
            add_test(NAME volume_contract COMMAND volume_checks)
        endif()
        if(menu_has_playback)
            add_executable(playback_checks Tests/playback.cpp)
            target_link_libraries(playback_checks PRIVATE menu_application)
            target_link_options(playback_checks PRIVATE -static)
            add_test(NAME playback_contract COMMAND playback_checks)
        endif()
        if(menu_has_load)
    add_executable(load_checks Tests/load.cpp)
    target_link_libraries(load_checks PRIVATE menu_application)
    target_compile_options(load_checks PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_link_options(load_checks PRIVATE -static)
    add_test(NAME load_contract COMMAND load_checks)
        endif()
        add_test(NAME round_front_panel_workflow COMMAND starter --front-panel-self-test)
        set_tests_properties(round_front_panel_workflow PROPERTIES TIMEOUT 20 PASS_REGULAR_EXPRESSION "PASS:")
        add_test(NAME round_menu_workflow COMMAND starter --interaction-self-test)
        set_tests_properties(round_menu_workflow PROPERTIES TIMEOUT 20 PASS_REGULAR_EXPRESSION "PASS:")
    endif()
else()
    set(hal_root ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/STM32L4xx_HAL_Driver)
    set(bsp_root ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/BSP/STM32L4R9I-Discovery)
    set(components ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/BSP/Components)
    set(hal_sources)
    foreach(module IN ITEMS hal hal_cortex hal_dma hal_dma_ex hal_dma2d hal_dsi hal_flash hal_flash_ex hal_flash_ramfunc hal_gfxmmu hal_gpio hal_i2c hal_i2c_ex hal_ltdc hal_ltdc_ex hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex hal_uart hal_uart_ex)
        list(APPEND hal_sources ${hal_root}/Src/stm32l4xx_${module}.c)
    endforeach()
    add_library(board_vendor STATIC ${hal_sources} ${bsp_root}/stm32l4r9i_discovery.c ${bsp_root}/stm32l4r9i_discovery_io.c
        ${components}/mfxstm32l152/mfxstm32l152.c ${components}/ft3x67/ft3x67.c Core/Src/system_stm32l4xx.c)
    target_include_directories(board_vendor PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cmake/hal)
    target_include_directories(board_vendor SYSTEM PUBLIC ${hal_root}/Inc ${hal_root}/Inc/Legacy
        ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS_6/CMSIS/Core/Include
        ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS/Device/ST/STM32L4xx/Include
        ${bsp_root} ${components}/ft3x67 ${components}/mfxstm32l152)
    target_compile_definitions(board_vendor PUBLIC USE_HAL_DRIVER STM32L4R9xx)
    target_link_libraries(starter_config INTERFACE board_vendor)
    if(menu_has_ui)
        target_sources(starter_application PRIVATE Platform/STM32/task_ui.cpp Platform/STM32/ui.cpp)
    else()
        target_sources(starter_application PRIVATE Application/no_ui.cpp)
    endif()
    add_executable(starter startup_stm32l4r9xx.s cmake/runtime/syscalls.c Board/system_clock.cpp
        Platform/STM32/main.cpp Devices/display_driver/display_driver.cpp
        Devices/display_driver/display_hal.c Devices/input_controller/input_controller.cpp)
    set_target_properties(starter PROPERTIES SUFFIX .elf)
    target_link_libraries(starter PRIVATE starter_application starter_rtos board_vendor)
    target_link_options(starter PRIVATE -T${CMAKE_CURRENT_SOURCE_DIR}/cmake/stm32l4r9-display.ld
        --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--print-memory-usage
        -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/starter.map)
    set_property(TARGET starter APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cmake/stm32l4r9-display.ld)
    add_custom_command(TARGET starter POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:starter> ${CMAKE_CURRENT_BINARY_DIR}/starter.bin
        COMMAND ${CMAKE_SIZE} $<TARGET_FILE:starter>)
endif()
