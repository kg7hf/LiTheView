cmake_minimum_required(VERSION 3.25)
project(H750Starter LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(STARTER_PLATFORM "Win32" CACHE STRING "Win32 or STM32")
set_property(CACHE STARTER_PLATFORM PROPERTY STRINGS Win32 STM32)
if(NOT STARTER_PLATFORM MATCHES "^(Win32|STM32)$")
    message(FATAL_ERROR "Select STARTER_PLATFORM=Win32 or STM32")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "CMSIS-RTOS2 requires a 32-bit build. Use the win32-debug i686 preset.")
endif()
set(RTOS ${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/FreeRTOS/Source)
set(CMSIS ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS_6/CMSIS)
set(HAL ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/STM32H7xx_HAL_Driver)
set(DEVICE ${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS/Device/ST/STM32H7xx)
foreach(required IN ITEMS "${RTOS}/tasks.c" "${CMSIS}/Core/Include/cmsis_gcc.h"
                          "${HAL}/Src/stm32h7xx_hal.c" "${DEVICE}/Include/stm32h750xx.h")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing pinned dependency ${required}. Run tools/init-platform.ps1 without recursion.")
    endif()
endforeach()
if(STARTER_PLATFORM STREQUAL "Win32")
    set(PORT ${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/FreeRTOS-Kernel/portable/MSVC-MingW)
else()
    set(PORT ${RTOS}/portable/GCC/ARM_CM4F)
endif()
add_library(starter_config INTERFACE)
target_include_directories(starter_config INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Platform/${STARTER_PLATFORM} ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(starter_config SYSTEM INTERFACE
    ${RTOS}/include ${RTOS}/CMSIS_RTOS_V2 ${CMSIS}/RTOS2/Include ${PORT})
if(STARTER_PLATFORM STREQUAL "STM32")
    target_include_directories(starter_config SYSTEM INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/Core/Inc ${HAL}/Inc ${HAL}/Inc/Legacy
        ${DEVICE}/Include ${CMSIS}/Core/Include)
    target_compile_definitions(starter_config INTERFACE USE_HAL_DRIVER USE_PWR_LDO_SUPPLY STM32H750xx)
endif()
# Both builds compile the same pinned kernel and CMSIS wrapper. Only port.c changes.
add_library(starter_rtos STATIC
    ${RTOS}/tasks.c ${RTOS}/queue.c ${RTOS}/list.c ${RTOS}/timers.c
    ${RTOS}/event_groups.c ${RTOS}/stream_buffer.c
    ${RTOS}/CMSIS_RTOS_V2/cmsis_os2.c ${PORT}/port.c)
target_link_libraries(starter_rtos PUBLIC starter_config)
# The application grows in place. Watch the export marker even when it does not
# exist yet, so a normal Ninja/CMake Tools build notices the first Generate UI.
# No cache option may keep an old blank build selected after generation.
unset(STARTER_WITH_UI CACHE)
file(GLOB starter_ui_exports CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/simulation.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/component.json")
file(GLOB starter_form_handlers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/form_handlers.hpp")
set(starter_has_ui FALSE)
if(starter_ui_exports)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${starter_ui_exports})
    set(ui_manifest_path "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/component.json")
    if(NOT EXISTS "${ui_manifest_path}")
        set(ui_manifest_path "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/simulation.json")
    endif()
    file(READ "${ui_manifest_path}" ui_manifest)
    string(JSON generation_status GET "${ui_manifest}" generationStatus)
    if(NOT generation_status STREQUAL "complete")
        message(FATAL_ERROR "The UI export is incomplete. Save the form and Generate UI before building.")
    endif()
    foreach(required IN ITEMS form.hpp product_config.hpp display.hpp resources.hpp resources.cmake)
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/${required}")
            message(FATAL_ERROR "UI export is missing ${required}. Save the form and Generate UI again.")
        endif()
    endforeach()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/form.hpp" generated_form)
    if(NOT generated_form MATCHES "kGeneratorContractVersion\\{2\\}")
        message(FATAL_ERROR "This form predates typed handlers. Open it with Designer 0.2.4+, save and Generate UI, then build the same preset again.")
    endif()
    set(starter_has_ui TRUE)
    message(STATUS "Application includes the generated UI")
else()
    message(STATUS "Application has no generated UI yet")
endif()
add_library(starter_application OBJECT Application/app_main.cpp Tasks/kernel_memory.cpp)
target_link_libraries(starter_application PUBLIC starter_config)
target_compile_options(starter_application PRIVATE -Wall -Wextra -Wconversion -Wshadow -Wold-style-cast
    -Wdouble-promotion -Werror -fno-exceptions -fno-rtti -fno-threadsafe-statics)
if(starter_has_ui)
    set(LITHEVIEW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LiTheView" CACHE PATH "Application dependency")
    if(NOT EXISTS "${LITHEVIEW_ROOT}/include/lithe/form_instance.hpp")
        message(FATAL_ERROR "Initialize/update the pinned LiTheView package with typed form support: tools/init-litheview.ps1")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tasks/UI/Generated/form.hpp")
        message(FATAL_ERROR "Create First Display with LiTheView: Add LiTheView UI, then Generate UI. See README.md.")
    endif()
    foreach(option IN ITEMS BUILD_RENDERING_FIXTURES BUILD_RENDERING_TESTS BUILD_WIN32 BUILD_TESTS BUILD_DESIGNER_PREVIEW
                            BUILD_STM32U545_PROBE ENABLE_EMBEDDED_BUDGET_GATE)
        set(LITHEVIEW_${option} OFF CACHE BOOL "Starter builds only the library" FORCE)
    endforeach()
    set(LITHEVIEW_DESTINATION_FORMAT RGB565 CACHE STRING "This board adapter uses native RGB565" FORCE)
    add_subdirectory("${LITHEVIEW_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/litheview" EXCLUDE_FROM_ALL)
    target_link_libraries(starter_application PUBLIC LitheView::Core)
    if(TARGET LitheView::CompactFont)
        target_link_libraries(starter_application PUBLIC LitheView::CompactFont)
    endif()
    if(starter_form_handlers)
        set(starter_form_include "Tasks/UI/form_handlers.hpp")
        set(starter_form_type "application::ui::Form")
    else()
        set(starter_form_include "Tasks/UI/Generated/product_config.hpp")
        set(starter_form_type "lithe::simulator::Form::Instance<>")
    endif()
    configure_file(cmake/starter_form.hpp.in generated/starter_form.hpp @ONLY)
    target_include_directories(starter_application PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_sources(starter_application PRIVATE Lessons/FirstDisplay/ui.cpp
        Lessons/FirstDisplay/${STARTER_PLATFORM}/display.cpp Lessons/FirstDisplay/${STARTER_PLATFORM}/input.cpp)
    include(Tasks/UI/Generated/resources.cmake)
    foreach(resource IN LISTS LITHEVIEW_SELECTED_RESOURCE_SOURCES)
        target_sources(starter_application PRIVATE "${resource}")
    endforeach()
else()
    target_sources(starter_application PRIVATE Application/no_ui.cpp)
endif()
if(STARTER_PLATFORM STREQUAL "Win32")
    add_executable(starter WIN32 Platform/Win32/main.cpp)
    include(cmake/front_panel.cmake)
    if(NOT starter_has_ui)
        target_sources(starter PRIVATE Platform/Win32/no_display.cpp)
    endif()
    target_link_libraries(starter PRIVATE starter_application starter_rtos user32 gdi32 winmm)
    target_compile_options(starter PRIVATE -Wall -Wextra -Wconversion -Wshadow -Wold-style-cast
        -Wdouble-promotion -Werror -fno-exceptions -fno-rtti)
    target_link_options(starter PRIVATE -static -Wl,--gc-sections)
    enable_testing()
    add_executable(starter_input_checks Tests/button_input.cpp)
    target_link_libraries(starter_input_checks PRIVATE starter_config)
    target_compile_options(starter_input_checks PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_link_options(starter_input_checks PRIVATE -static)
    add_test(NAME button_input_contract COMMAND starter_input_checks)
    add_test(NAME one_application_task COMMAND starter --self-test)
    set_tests_properties(one_application_task PROPERTIES TIMEOUT 20 PASS_REGULAR_EXPRESSION "PASS:")
else()
    include(cmake/check_cube.cmake)
    include(cmake/hal_sources.cmake)
    add_library(starter_hal STATIC ${STARTER_HAL_SOURCES})
    target_link_libraries(starter_hal PUBLIC starter_config)
    add_executable(starter
        Core/Src/main.c Core/Src/stm32h7xx_it.c Core/Src/stm32h7xx_hal_msp.c
        Core/Src/stm32h7xx_hal_timebase_tim.c Core/Src/system_stm32h7xx.c
        Core/Src/syscalls.c Core/Src/sysmem.c startup_stm32h750xx.s
        Platform/STM32/indicator.cpp)
    set_target_properties(starter PROPERTIES SUFFIX .elf)
    target_link_libraries(starter PRIVATE starter_application starter_rtos starter_hal)
    target_compile_options(starter PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wconversion;-Wshadow;-Wold-style-cast;-Wdouble-promotion;-Werror>)
    target_link_options(starter PRIVATE
        "-Wl,--defsym=__starter_framebuffer_bytes=$<IF:$<BOOL:${starter_has_ui}>,522240,0>"
        -T${CMAKE_CURRENT_SOURCE_DIR}/STM32H750xx_FLASH.ld
        --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections
        -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/starter.map -Wl,--print-memory-usage)
    set_property(TARGET starter APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/STM32H750xx_FLASH.ld)
    add_custom_command(TARGET starter POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:starter> ${CMAKE_CURRENT_BINARY_DIR}/starter.bin
        COMMAND ${CMAKE_SIZE} $<TARGET_FILE:starter>)
endif()
