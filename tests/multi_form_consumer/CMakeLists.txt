cmake_minimum_required(VERSION 3.25)
project(MultiFormConsumer LANGUAGES CXX)
if(NOT DEFINED GENERATED_ROOT)
    message(FATAL_ERROR "Run Designer.Core.Tests, then pass its build/multi-form-tests/latest-consumer.txt path as GENERATED_ROOT.")
endif()
set(LITHEVIEW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LITHEVIEW_BUILD_WIN32 OFF CACHE BOOL "" FORCE)
set(LITHEVIEW_BUILD_RENDERING_TESTS OFF CACHE BOOL "" FORCE)
set(LITHEVIEW_BUILD_RENDERING_FIXTURES OFF CACHE BOOL "" FORCE)
set(LITHEVIEW_BUILD_DESIGNER_PREVIEW OFF CACHE BOOL "" FORCE)
set(LITHEVIEW_DESTINATION_FORMAT RGB565 CACHE STRING "" FORCE)
add_subdirectory(../.. litheview EXCLUDE_FROM_ALL)
add_subdirectory("${GENERATED_ROOT}" forms/Legacy)
foreach(form IN ITEMS Settings Foo FooAssets)
    add_subdirectory("${GENERATED_ROOT}/${form}" "forms/${form}")
endforeach()
add_executable(multi_form_consumer main.cpp)
target_include_directories(multi_form_consumer PRIVATE "${GENERATED_ROOT}")
target_link_libraries(multi_form_consumer PRIVATE LitheView::GeneratedComponent
    LitheView::FormSettings LitheView::FormFoo LitheView::FormFooAssets LitheView::CompactFont)
enable_testing()
add_test(NAME generated_multi_form_consumer COMMAND multi_form_consumer)
