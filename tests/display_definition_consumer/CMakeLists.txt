cmake_minimum_required(VERSION 3.25)
project(LitheViewDisplayDefinitionConsumer LANGUAGES CXX)
find_package(LitheView CONFIG REQUIRED)
set(GENERATED_DIRECTORY "" CACHE PATH "Headless display export directory")
set(DISPLAY_EXAMPLE "" CACHE STRING "Checked display example basename")
set(expected_shape rectangle)
set(expected_address pixels)
set(expected_safe_x 0)
set(expected_safe_y 0)
set(expected_resource gallopAlpha1)
if(DISPLAY_EXAMPLE STREQUAL "rk043fn48h-rgb565")
    set(expected_format rgb565)
    set(expected_width 480)
    set(expected_height 272)
elseif(DISPLAY_EXAMPLE STREQUAL "rk043fn48h-xrgb8888")
    set(expected_format xrgb8888)
    set(expected_width 480)
    set(expected_height 272)
elseif(DISPLAY_EXAMPLE STREQUAL "round-240-rgb565")
    set(expected_format rgb565)
    set(expected_width 240)
    set(expected_height 240)
    set(expected_shape circle)
    set(expected_safe_x 36)
    set(expected_safe_y 36)
elseif(DISPLAY_EXAMPLE STREQUAL "synthetic-r2")
    set(expected_format r2Packed2x2)
    set(expected_width 512)
    set(expected_height 384)
    set(expected_address emitters)
    set(expected_resource gallopR2)
elseif(DISPLAY_EXAMPLE STREQUAL "synthetic-red-cell")
    set(expected_format rrrr2222)
    set(expected_width 256)
    set(expected_height 192)
    set(expected_address cells)
    set(expected_resource gallopR2)
elseif(DISPLAY_EXAMPLE STREQUAL "synthetic-red-yellow")
    set(expected_format ryry2222)
    set(expected_width 256)
    set(expected_height 192)
    set(expected_address cells)
    set(expected_resource gallopR2)
else()
    message(FATAL_ERROR "DISPLAY_EXAMPLE must name a checked display definition")
endif()
math(EXPR expected_safe_width "${expected_width} - 2 * ${expected_safe_x}")
math(EXPR expected_safe_height "${expected_height} - 2 * ${expected_safe_y}")
configure_file(expected.hpp.in generated/expected.hpp @ONLY)
include("${GENERATED_DIRECTORY}/resources.cmake")
add_executable(litheview_display_definition_consumer main.cpp ${LITHEVIEW_SELECTED_RESOURCE_SOURCES})
target_include_directories(litheview_display_definition_consumer PRIVATE
    "${GENERATED_DIRECTORY}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(litheview_display_definition_consumer PRIVATE LitheView::Core LitheView::CompactFont)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(litheview_display_definition_consumer PRIVATE
        -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror
        -fno-exceptions -fno-rtti)
endif()
enable_testing()
add_test(NAME litheview_display_definition_consumer
    COMMAND litheview_display_definition_consumer "${CMAKE_CURRENT_BINARY_DIR}/frame.bin")
